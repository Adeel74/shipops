  1302→25. Order Creation Worker
  1303→async function processShopifyOrder(payload) {
  1304→
  1305→  const store = await getStoreFromWebhook(payload);
  1306→
  1307→  const customer = await upsertCustomer({
  1308→    organizationId: store.organizationId,
  1309→    shopifyCustomerId: payload.customer?.id,
  1310→    firstName: payload.customer?.first_name,
  1311→    lastName: payload.customer?.last_name,
  1312→    phone: payload.customer?.phone,
  1313→    email: payload.customer?.email
  1314→  });
  1315→
  1316→  const order = await createOrUpdateOrder({
  1317→    organizationId: store.organizationId,
  1318→    storeId: store.id,
  1319→    customerId: customer.id,
  1320→    shopifyOrderId: String(payload.id),
  1321→    orderNumber: payload.order_number,
  1322→    totalAmount: payload.total_price,
  1323→    isCod: detectCOD(payload)
  1324→  });
  1325→
  1326→  await createOrderItems(order, payload);
  1327→
  1328→  await runAutomations({
  1329→    event: "ORDER_CREATED",
  1330→    orderId: order.id
  1331→  });
  1332→
  1333→  return order;
  1334→}
